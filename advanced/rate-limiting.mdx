## Overview

Rate limiting controls the number of requests that can be made to the WhatsApp API within a specific time period. This prevents abuse, ensures fair usage, and maintains service stability for all users.

## WhatsApp Rate Limits

### Message Limits

| Type | Limit | Period | Notes |
|------|-------|--------|-------|
| **Text Messages** | 1,000 | Per day | Per phone number |
| **Media Messages** | 100 | Per hour | Images, videos, documents |
| **Group Messages** | 250 | Per hour | Messages to groups |
| **Bulk Messages** | 80 | Per hour | Messages to multiple recipients |

### API Request Limits

| Endpoint Category | Limit | Period | Scope |
|-------------------|-------|--------|-------|
| **Send Messages** | 100 | Per minute | Per API instance |
| **User Operations** | 200 | Per minute | Per API instance |
| **Group Management** | 50 | Per minute | Per API instance |
| **Media Upload** | 20 | Per minute | Per API instance |

## Rate Limit Headers

The API includes rate limit information in response headers:

```http
HTTP/1.1 200 OK
X-RateLimit-Limit: 100
X-RateLimit-Remaining: 95
X-RateLimit-Reset: 1640995200
X-RateLimit-Retry-After: 60
Content-Type: application/json
```

### Header Descriptions

- `X-RateLimit-Limit`: Maximum requests allowed in the time window
- `X-RateLimit-Remaining`: Requests remaining in current window
- `X-RateLimit-Reset`: Unix timestamp when the limit resets
- `X-RateLimit-Retry-After`: Seconds to wait before making another request

```javascript JavaScript
async function sendMessageWithRateLimit(phone, message) {
  try {
    const response = await fetch('http://localhost:3000/send/message', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ phone, message })
    });
    
    // Check rate limit headers
    const remaining = parseInt(response.headers.get('X-RateLimit-Remaining'));
    const resetTime = parseInt(response.headers.get('X-RateLimit-Reset'));
    
    if (response.status === 429) {
      const retryAfter = parseInt(response.headers.get('X-RateLimit-Retry-After'));
      console.warn(`Rate limit exceeded. Retry after ${retryAfter} seconds`);
      
      // Wait and retry
      await sleep(retryAfter * 1000);
      return await sendMessageWithRateLimit(phone, message);
    }
    
    if (remaining < 10) {
      console.warn(`Rate limit warning: ${remaining} requests remaining`);
    }
    
    return await response.json();
  } catch (error) {
    console.error('Failed to send message:', error);
    throw error;
  }
}

function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}
```

```go Go
        time.Sleep(time.Duration(retryAfter) * time.Second)

        fmt.Printf("Rate limit warning: %d requests remaining\n", remaining)
    }
```

### Advanced Rate Limiting

```javascript
class RateLimiter {
  constructor(options = {}) {
    this.limits = options.limits || {
      requests: { max: 100, window: 60000 }, // 100 requests per minute
      messages: { max: 1000, window: 86400000 } // 1000 messages per day
    };
    
    this.counters = new Map();
    this.queues = new Map();
  }
  
  async checkLimit(key, type = 'requests') {
    const limit = this.limits[type];
    const now = Date.now();
    const windowStart = now - limit.window;
    
    // Clean old entries
    if (this.counters.has(key)) {
      this.counters.get(key)[type] = this.counters.get(key)[type]
        .filter(timestamp => timestamp > windowStart);
    } else {
      this.counters.set(key, { requests: [], messages: [] });
    }
    
    const counter = this.counters.get(key)[type];
    
    if (counter.length >= limit.max) {
      const oldestRequest = Math.min(...counter);
      const waitTime = oldestRequest + limit.window - now;
      
      if (waitTime > 0) {
        throw new RateLimitError(`Rate limit exceeded. Wait ${Math.ceil(waitTime / 1000)} seconds`, waitTime);
      }
    }
    
    // Add current request
    counter.push(now);
    
    return {
      remaining: limit.max - counter.length,
      resetTime: windowStart + limit.window,
      limit: limit.max
    };
  }
  
  async executeWithLimit(key, fn, type = 'requests') {
    try {
      const rateLimitInfo = await this.checkLimit(key, type);
      console.log(`Rate limit info: ${rateLimitInfo.remaining}/${rateLimitInfo.limit} remaining`);
      
      return await fn();
    } catch (error) {
      if (error instanceof RateLimitError) {
        console.warn(`Rate limited for ${Math.ceil(error.waitTime / 1000)} seconds`);
        await this.sleep(error.waitTime);
        return await this.executeWithLimit(key, fn, type);
      }
      throw error;
    }
  }
  
  async queueRequest(key, fn, type = 'requests') {
    if (!this.queues.has(key)) {
      this.queues.set(key, []);
    }
    
    return new Promise((resolve, reject) => {
      this.queues.get(key).push({ fn, type, resolve, reject });
      this.processQueue(key);
    });
  }
  
  async processQueue(key) {
    const queue = this.queues.get(key);
    if (!queue || queue.length === 0) return;
    
    const { fn, type, resolve, reject } = queue.shift();
    
    try {
      const result = await this.executeWithLimit(key, fn, type);
      resolve(result);
    } catch (error) {
      reject(error);
    }
    
    // Process next item after delay
    setTimeout(() => this.processQueue(key), 1000);
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

class RateLimitError extends Error {
  constructor(message, waitTime) {
    super(message);
    this.name = 'RateLimitError';
    this.waitTime = waitTime;
  }
}

// Usage
const rateLimiter = new RateLimiter({
  limits: {
    requests: { max: 100, window: 60000 }, // 100/minute
    messages: { max: 1000, window: 86400000 } // 1000/day
  }
});

async function sendMessage(phone, message) {
  return await rateLimiter.executeWithLimit('api', async () => {
    const response = await fetch('http://localhost:3000/send/message', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ phone, message })
    });
    return await response.json();
  }, 'messages');
}
```

## Batch Processing

### Message Queue with Rate Limiting

```javascript
class MessageQueue {
  constructor(options = {}) {
    this.queue = [];
    this.processing = false;
    this.rateLimit = options.rateLimit || { max: 60, window: 60000 }; // 60/minute
    this.concurrency = options.concurrency || 1;
    this.delay = options.delay || 1000; // 1 second between messages
  }
  
  add(phone, message, priority = 0) {
    this.queue.push({
      phone,
      message,
      priority,
      timestamp: Date.now(),
      attempts: 0
    });
    
    // Sort by priority (higher number = higher priority)
    this.queue.sort((a, b) => b.priority - a.priority);
    
    if (!this.processing) {
      this.process();
    }
  }
  
  async process() {
    if (this.processing) return;
    
    this.processing = true;
    
    while (this.queue.length > 0) {
      const batch = this.queue.splice(0, this.concurrency);
      
      const promises = batch.map(async (item) => {
        try {
          await this.sendMessage(item);
          console.log(`✅ Message sent to ${item.phone}`);
        } catch (error) {
          console.error(`❌ Failed to send to ${item.phone}:`, error.message);
          
          // Retry logic
          if (item.attempts < 3) {
            item.attempts++;
            this.queue.unshift(item); // Add back to front of queue
          }
        }
      });
      
      await Promise.allSettled(promises);
      
      // Wait between batches
      if (this.queue.length > 0) {
        await this.sleep(this.delay);
      }
    }
    
    this.processing = false;
  }
  
  async sendMessage(item) {
    const response = await fetch('http://localhost:3000/send/message', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        phone: item.phone,
        message: item.message
      })
    });
    
    if (response.status === 429) {
      const retryAfter = parseInt(response.headers.get('X-RateLimit-Retry-After')) || 60;
      throw new Error(`Rate limited. Retry after ${retryAfter} seconds`);
    }
    
    if (!response.ok) {
      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
    }
    
    return await response.json();
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
  
  getStatus() {
    return {
      queueLength: this.queue.length,
      processing: this.processing
    };
  }
}

// Usage
const messageQueue = new MessageQueue({
  rateLimit: { max: 100, window: 60000 },
  concurrency: 5,
  delay: 500
});

// Add messages to queue
const recipients = [
  '6289685028129@s.whatsapp.net',
  '6289685028130@s.whatsapp.net',
  '6289685028131@s.whatsapp.net'
];

recipients.forEach(phone => {
  messageQueue.add(phone, 'Hello! This is a bulk message.', 1);
});

// Monitor queue status
setInterval(() => {
  const status = messageQueue.getStatus();
  console.log(`Queue status: ${status.queueLength} pending, processing: ${status.processing}`);
}, 5000);
```

## Exponential Backoff

```javascript
class ExponentialBackoff {
  constructor(options = {}) {
    this.baseDelay = options.baseDelay || 1000; // 1 second
    this.maxDelay = options.maxDelay || 300000; // 5 minutes
    this.multiplier = options.multiplier || 2;
    this.jitter = options.jitter || true;
    this.maxRetries = options.maxRetries || 5;
  }
  
  async executeWithBackoff(fn, attempt = 1) {
    try {
      return await fn();
    } catch (error) {
      if (attempt >= this.maxRetries) {
        throw new Error(`Max retries (${this.maxRetries}) exceeded: ${error.message}`);
      }
      
      if (error.status === 429 || error.message.includes('rate limit')) {
        const delay = this.calculateDelay(attempt);
        console.warn(`Attempt ${attempt} failed, retrying in ${delay}ms:`, error.message);
        
        await this.sleep(delay);
        return await this.executeWithBackoff(fn, attempt + 1);
      }
      
      throw error;
    }
  }
  
  calculateDelay(attempt) {
    let delay = this.baseDelay * Math.pow(this.multiplier, attempt - 1);
    delay = Math.min(delay, this.maxDelay);
    
    if (this.jitter) {
      // Add random jitter (±25%)
      const jitterRange = delay * 0.25;
      delay += (Math.random() * 2 - 1) * jitterRange;
    }
    
    return Math.max(delay, 0);
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

// Usage
const backoff = new ExponentialBackoff({
  baseDelay: 1000,
  maxDelay: 60000,
  multiplier: 2,
  maxRetries: 5
});

async function sendMessageWithBackoff(phone, message) {
  return await backoff.executeWithBackoff(async () => {
    const response = await fetch('http://localhost:3000/send/message', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ phone, message })
    });
    
    if (response.status === 429) {
      throw { status: 429, message: 'Rate limit exceeded' };
    }
    
    if (!response.ok) {
      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
    }
    
    return await response.json();
  });
}
```

## Monitoring Rate Limits

```javascript
class RateLimitMonitor {
  constructor() {
    this.metrics = {
      requests: 0,
      rateLimited: 0,
      errors: 0,
      averageResponseTime: 0
    };
    
    this.history = [];
  }
  
  recordRequest(responseTime, rateLimited = false, error = false) {
    this.metrics.requests++;
    
    if (rateLimited) this.metrics.rateLimited++;
    if (error) this.metrics.errors++;
    
    // Update average response time
    this.metrics.averageResponseTime = 
      (this.metrics.averageResponseTime * (this.metrics.requests - 1) + responseTime) / 
      this.metrics.requests;
    
    // Store history for analysis
    this.history.push({
      timestamp: Date.now(),
      responseTime,
      rateLimited,
      error
    });
    
    // Keep only last 1000 entries
    if (this.history.length > 1000) {
      this.history = this.history.slice(-1000);
    }
  }
  
  getMetrics() {
    const now = Date.now();
    const oneHourAgo = now - 3600000;
    
    const recentHistory = this.history.filter(entry => entry.timestamp > oneHourAgo);
    
    return {
      ...this.metrics,
      rateLimitRate: this.metrics.rateLimited / this.metrics.requests,
      errorRate: this.metrics.errors / this.metrics.requests,
      requestsLastHour: recentHistory.length,
      rateLimitedLastHour: recentHistory.filter(e => e.rateLimited).length
    };
  }
  
  generateReport() {
    const metrics = this.getMetrics();
    
    console.log('=== Rate Limit Monitor Report ===');
    console.log(`Total Requests: ${metrics.requests}`);
    console.log(`Rate Limited: ${metrics.rateLimited} (${(metrics.rateLimitRate * 100).toFixed(2)}%)`);
    console.log(`Errors: ${metrics.errors} (${(metrics.errorRate * 100).toFixed(2)}%)`);
    console.log(`Average Response Time: ${metrics.averageResponseTime.toFixed(2)}ms`);
    console.log(`Requests Last Hour: ${metrics.requestsLastHour}`);
    console.log(`Rate Limited Last Hour: ${metrics.rateLimitedLastHour}`);
    console.log('===============================');
  }
}

// Global monitor instance
const rateLimitMonitor = new RateLimitMonitor();

// Wrap API calls with monitoring
async function monitoredApiCall(fn) {
  const startTime = Date.now();
  let rateLimited = false;
  let error = false;
  
  try {
    const result = await fn();
    return result;
  } catch (err) {
    error = true;
    if (err.status === 429 || err.message.includes('rate limit')) {
      rateLimited = true;
    }
    throw err;
  } finally {
    const responseTime = Date.now() - startTime;
    rateLimitMonitor.recordRequest(responseTime, rateLimited, error);
  }
}

// Generate reports every hour
setInterval(() => {
  rateLimitMonitor.generateReport();
}, 3600000);
```
