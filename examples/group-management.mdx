### Advanced Group Creation

```javascript
class GroupManager {
  constructor(apiUrl) {
    this.apiUrl = apiUrl;
  }
  
  async createGroup(title, participants, options = {}) {
    try {
      const payload = {
        title,
        participants,
        ...options
      };
      
      const response = await fetch(`${this.apiUrl}/group`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(payload)
      });
      
      const result = await response.json();
      
      if (result.code === 'SUCCESS') {
        const groupId = result.results.group_id;
        
        // Configure group settings after creation
        if (options.description) {
          await this.setGroupDescription(groupId, options.description);
        }
        
        if (options.photo) {
          await this.setGroupPhoto(groupId, options.photo);
        }
        
        if (options.settings) {
          await this.configureGroupSettings(groupId, options.settings);
        }
        
        return {
          groupId,
          title,
          participants: participants.length
        };
      } else {
        throw new Error(result.message);
      }
    } catch (error) {
      console.error('Failed to create group:', error);
      throw error;
    }
  }
  
  async createGroupWithValidation(title, participants, options = {}) {
    // Validate participants before creating group
    const validParticipants = [];
    
    for (const phone of participants) {
      const isValid = await this.validatePhoneNumber(phone);
      if (isValid) {
        validParticipants.push(phone);
      } else {
        console.warn(`Invalid phone number: ${phone}`);
      }
    }
    
    if (validParticipants.length < 1) {
      throw new Error('At least one valid participant is required');
    }
    
    return await this.createGroup(title, validParticipants, options);
  }
  
  async validatePhoneNumber(phone) {
    try {
      const response = await fetch(`${this.apiUrl}/user/check?phone=${phone}`);
      const result = await response.json();
      return result.code === 'SUCCESS' && result.results.is_on_whatsapp;
    } catch (error) {
      console.error(`Failed to validate ${phone}:`, error);
      return false;
    }
  }
}
```

### Group Analytics

```javascript
class GroupAnalytics {
  constructor(apiUrl) {
    this.apiUrl = apiUrl;
  }
  
  async getGroupStats(groupId) {
    try {
      const groupInfo = await this.getGroupInfo(groupId);
      const messages = await this.getGroupMessages(groupId, { limit: 100 });
      
      const stats = {
        groupId,
        name: groupInfo.Name,
        totalParticipants: groupInfo.Participants.length,
        adminCount: groupInfo.Participants.filter(p => p.IsAdmin).length,
        memberCount: groupInfo.Participants.filter(p => !p.IsAdmin).length,
        isLocked: groupInfo.IsLocked,
        isAnnounce: groupInfo.IsAnnounce,
        createdAt: groupInfo.GroupCreated,
        recentActivity: this.analyzeRecentActivity(messages)
      };
      
      return stats;
    } catch (error) {
      console.error('Failed to get group stats:', error);
      throw error;
    }
  }
  
  async getGroupInfo(groupId) {
    const response = await fetch(`${this.apiUrl}/group/info?group_id=${encodeURIComponent(groupId)}`);
    const result = await response.json();
    
    if (result.code === 'SUCCESS') {
      return result.results;
    } else {
      throw new Error(result.message);
    }
  }
  
  async getGroupMessages(groupId, options = {}) {
    const params = new URLSearchParams({
      limit: options.limit || 50,
      offset: options.offset || 0
    });
    
    const response = await fetch(`${this.apiUrl}/chat/${encodeURIComponent(groupId)}/messages?${params}`);
    const result = await response.json();
    
    if (result.code === 'SUCCESS') {
      return result.results.data;
    } else {
      throw new Error(result.message);
    }
  }
  
  analyzeRecentActivity(messages) {
    const now = new Date();
    const oneDayAgo = new Date(now - 24 * 60 * 60 * 1000);
    const oneWeekAgo = new Date(now - 7 * 24 * 60 * 60 * 1000);
    
    const recentMessages = messages.filter(msg => 
      new Date(msg.timestamp) > oneDayAgo
    );
    
    const weeklyMessages = messages.filter(msg => 
      new Date(msg.timestamp) > oneWeekAgo
    );
    
    return {
      messagesLast24h: recentMessages.length,
      messagesLastWeek: weeklyMessages.length,
      activeUsersLast24h: new Set(recentMessages.map(msg => msg.sender_jid)).size,
      activeUsersLastWeek: new Set(weeklyMessages.map(msg => msg.sender_jid)).size
    };
  }
}

console.log('Group statistics:', stats);
```

### Group Broadcast

```javascript
class GroupBroadcaster {
  constructor(apiUrl) {
    this.apiUrl = apiUrl;
  }
  
  async broadcastToMultipleGroups(groupIds, message, options = {}) {
    const results = [];
    const delay = options.delay || 1000; // 1 second delay between messages
    
    for (const groupId of groupIds) {
      try {
        const result = await this.sendGroupMessage(groupId, message);
        results.push({ groupId, success: true, result });
        
        // Add delay to avoid rate limiting
        if (groupIds.indexOf(groupId) < groupIds.length - 1) {
          await this.sleep(delay);
        }
      } catch (error) {
        console.error(`Failed to send to group ${groupId}:`, error);
        results.push({ groupId, success: false, error: error.message });
      }
    }
    
    return results;
  }
  
  async sendGroupMessage(groupId, message) {
    const response = await fetch(`${this.apiUrl}/send/message`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        phone: groupId,
        message
      })
    });
    
    return await response.json();
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

```javascript
function handleGroupEvent(event) {
  const { type, groupId, participant, data } = event;
  
  switch (type) {
    case 'participant_added':
      handleParticipantAdded(groupId, participant, data);
      break;
    case 'participant_removed':
      handleParticipantRemoved(groupId, participant, data);
      break;
    case 'participant_promoted':
      handleParticipantPromoted(groupId, participant);
      break;
    case 'participant_demoted':
      handleParticipantDemoted(groupId, participant);
      break;
    case 'group_name_changed':
      handleGroupNameChanged(groupId, data.oldName, data.newName);
      break;
    case 'group_description_changed':
      handleGroupDescriptionChanged(groupId, data.newDescription);
      break;
    default:
      console.log('Unknown group event:', type);
  }
}
```

```javascript
      await fetch(`${this.apiUrl}/message/${messageId}/delete`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({})
      });
    } catch (error) {
      console.error('Failed to delete message:', error);
    }
  }
  
  async sendWarning(groupId, userId, violations, warningCount) {
    const violationText = violations.join(', ');
    const warningMessage = `⚠️ Warning ${warningCount}/3 for @${userId}\n\nViolation: ${violationText}\n\nPlease follow group rules.`;
    
    await sendGroupMessage(groupId, warningMessage);
  }
  
  async removeParticipant(groupId, userId) {
    try {
      await fetch(`${this.apiUrl}/group/participants/remove`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          group_id: groupId,
          participants: [userId.replace('@s.whatsapp.net', '')]
        })
      });
    } catch (error) {
      console.error('Failed to remove participant:', error);
    }
  }
  
  async notifyRemoval(groupId, userId) {
    const message = `🚫 User @${userId} has been removed from the group for repeated violations.`;
    await sendGroupMessage(groupId, message);
  }
}

// Usage
const moderator = new GroupModerator('http://localhost:3000');

// Handle incoming messages
function handleIncomingMessage(message) {
  if (message.isGroup) {
    moderator.moderateMessage(message);
  }
}
```
