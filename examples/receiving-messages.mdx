## Overview

There are two main approaches to receive WhatsApp messages with the API:

1. **Webhooks** (Recommended) - Real-time message delivery via HTTP callbacks
2. **Polling** - Periodically check for new messages via API calls

This guide covers both methods and shows you how to handle different types of incoming messages.

## Webhooks (Recommended)

Webhooks provide real-time message delivery by sending HTTP POST requests to your server whenever a new message arrives.

### Setting Up Webhooks

Configure your webhook endpoint in the API:

<CodeGroup>
```bash cURL
curl -X POST http://localhost:3000/webhook/set \
  -H "Content-Type: application/json" \
  -d '{
    "url": "https://your-server.com/webhook",
    "events": ["message", "message_ack", "group_update"]
  }'
```

```javascript JavaScript

  method: 'POST',

  },

    events: ['message', 'message_ack', 'group_update']
  })
});
```
</CodeGroup>

### Webhook Payload Structure

When a message is received, your webhook endpoint will receive a POST request with this structure:

```json
{
  "event": "message",
  "data": {
    "id": "3EB0B430B6F8F1D0E053AC120E0A9E5C",
    "from": "6289685028129@s.whatsapp.net",
    "to": "6289685028130@s.whatsapp.net",
    "timestamp": "2024-01-15T10:30:00Z",
    "type": "text",
    "body": "Hello! This is an incoming message.",
    "fromMe": false,
    "hasMedia": false,
    "participant": null,
    "isGroup": false,
    "groupId": null
  }
}
```

    ```go
    package main
    
    import (
        "bytes"
        "encoding/json"
        "fmt"
        "log"
        "net/http"
    )
    
    type WebhookPayload struct {
        Event string      `json:"event"`
        Data  MessageData `json:"data"`
    }
    
    type MessageData struct {
        ID        string `json:"id"`
        From      string `json:"from"`
        To        string `json:"to"`
        Type      string `json:"type"`
        Body      string `json:"body"`
        FromMe    bool   `json:"fromMe"`
        HasMedia  bool   `json:"hasMedia"`
        IsGroup   bool   `json:"isGroup"`
    }
    
    func webhookHandler(w http.ResponseWriter, r *http.Request) {
        var payload WebhookPayload
        
        if err := json.NewDecoder(r.Body).Decode(&payload); err != nil {
            http.Error(w, "Invalid JSON", http.StatusBadRequest)
            return
        }
        
        fmt.Printf("Received webhook: %s\n", payload.Event)
        
        switch payload.Event {
        case "message":
            handleIncomingMessage(payload.Data)
        case "message_ack":
            handleMessageAck(payload.Data)
        case "group_update":
            handleGroupUpdate(payload.Data)
        }
        
        w.WriteHeader(http.StatusOK)
        json.NewEncoder(w).Encode(map[string]string{"status": "received"})
    }
    
    func handleIncomingMessage(message MessageData) {
        fmt.Printf("New message from %s: %s\n", message.From, message.Body)
        
        // Auto-reply example
        if message.Body == "hello" {
            sendReply(message.From, "Hello! How can I help you?")
        }
    }
    
    func sendReply(phone, message string) {
        payload := map[string]string{
            "phone":   phone,
            "message": message,
        }
        
        jsonData, _ := json.Marshal(payload)
        
        resp, err := http.Post("http://localhost:3000/send/message", 
            "application/json", bytes.NewBuffer(jsonData))
        if err != nil {
            fmt.Printf("Failed to send reply: %v\n", err)
            return
        }
        defer resp.Body.Close()
        
        fmt.Println("Reply sent successfully")
    }
    
    func main() {
        http.HandleFunc("/webhook", webhookHandler)
        
        fmt.Println("Webhook server running on port 8080")
        log.Fatal(http.ListenAndServe(":8080", nil))
    ```

## Message Types

Handle different types of incoming messages:

### Text Messages

```javascript
function handleIncomingMessage(message) {
  if (message.type === 'text') {
    console.log(`Text message: ${message.body}`);
    
    // Command handling
    if (message.body.startsWith('/')) {
      handleCommand(message);
    } else {
      handleRegularMessage(message);
    }
  }
}

function handleCommand(message) {
  const command = message.body.split(' ')[0].toLowerCase();
  const args = message.body.split(' ').slice(1);
  
  switch (command) {
    case '/help':
      sendReply(message.from, 'Available commands:\n/help - Show this help\n/time - Get current time\n/weather - Get weather info');
      break;
    case '/time':
      sendReply(message.from, `Current time: ${new Date().toLocaleString()}`);
      break;
    case '/weather':
      const city = args.join(' ') || 'Jakarta';
      // Integrate with weather API
      sendReply(message.from, `Weather in ${city}: Sunny, 28°C`);
      break;
    default:
      sendReply(message.from, 'Unknown command. Type /help for available commands.');
  }
}
```

### Group Messages

```javascript
function handleGroupMessage(message) {
  if (message.isGroup) {
    console.log(`Group message in ${message.groupId} from ${message.participant}: ${message.body}`);
    
    // Handle mentions
    if (message.body.includes('@everyone')) {
      sendGroupReply(message.groupId, 'Hello everyone! 👋');
    }
    
    // Moderation
    if (containsInappropriateContent(message.body)) {
      // Delete message and warn user
      deleteMessage(message.id);
      sendPrivateMessage(message.participant, 'Please keep the conversation respectful.');
    }
  }
}

function containsInappropriateContent(text) {
  const bannedWords = ['spam', 'scam', 'inappropriate'];
  return bannedWords.some(word => text.toLowerCase().includes(word));
}
```

## Polling Method

If webhooks are not suitable for your setup, you can poll for new messages:

### Basic Polling

```javascript
class MessagePoller {
  constructor(apiUrl, pollInterval = 5000) {
    this.apiUrl = apiUrl;
    this.pollInterval = pollInterval;
    this.lastMessageId = null;
    this.isPolling = false;
  }
  
  start() {
    if (this.isPolling) return;
    
    this.isPolling = true;
    this.poll();
  }
  
  stop() {
    this.isPolling = false;
  }
  
  async poll() {
    while (this.isPolling) {
      try {
        const messages = await this.fetchNewMessages();
        
        for (const message of messages) {
          this.handleMessage(message);
        }
        
        await this.sleep(this.pollInterval);
      } catch (error) {
        console.error('Polling error:', error);
        await this.sleep(this.pollInterval * 2); // Wait longer on error
      }
    }
  }
  
  async fetchNewMessages() {
    const url = this.lastMessageId 
      ? `${this.apiUrl}/messages?after=${this.lastMessageId}`
      : `${this.apiUrl}/messages?limit=10`;
    
    const response = await fetch(url);
    const data = await response.json();
    
    if (data.code === 'SUCCESS' && data.results.length > 0) {
      this.lastMessageId = data.results[data.results.length - 1].id;
      return data.results;
    }
    
    return [];
  }
  
  handleMessage(message) {
    console.log('New message:', message);
    // Process message here
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

// Usage
const poller = new MessagePoller('http://localhost:3000');
poller.start();

// Stop polling when needed
// poller.stop();
```

### Advanced Polling with State Management

```javascript
class AdvancedMessagePoller {
  constructor(apiUrl, options = {}) {
    this.apiUrl = apiUrl;
    this.pollInterval = options.pollInterval || 5000;
    this.maxRetries = options.maxRetries || 3;
    this.processedMessages = new Set();
    this.isPolling = false;
  }
  
  async pollChatMessages(chatJid, limit = 50) {
    try {
      const response = await fetch(
        `${this.apiUrl}/chat/${encodeURIComponent(chatJid)}/messages?limit=${limit}`
      );
      const data = await response.json();
      
      if (data.code === 'SUCCESS') {
        return data.results.filter(msg => !this.processedMessages.has(msg.id));
      }
      
      return [];
    } catch (error) {
      console.error('Failed to fetch chat messages:', error);
      return [];
    }
  }
  
  async pollAllChats() {
    try {
      // Get list of all chats
      const chatsResponse = await fetch(`${this.apiUrl}/chats`);
      const chatsData = await chatsResponse.json();
      
      if (chatsData.code === 'SUCCESS') {
        for (const chat of chatsData.results.data) {
          const newMessages = await this.pollChatMessages(chat.jid, 10);
          
          for (const message of newMessages) {
            this.handleMessage(message);
            this.processedMessages.add(message.id);
          }
        }
      }
    } catch (error) {
      console.error('Failed to poll all chats:', error);
    }
  }
  
  handleMessage(message) {
    console.log(`New message in ${message.chat_jid}: ${message.content}`);
    
    // Route to appropriate handler
    if (message.media_type) {
      this.handleMediaMessage(message);
    } else {
      this.handleTextMessage(message);
    }
  }
  
  handleTextMessage(message) {
    // Process text message
    if (message.content.startsWith('/')) {
      this.handleCommand(message);
    }
  }
  
  handleMediaMessage(message) {
    console.log(`Media message: ${message.media_type}, Size: ${message.file_length}`);
  }
  
  handleCommand(message) {
    const command = message.content.split(' ')[0];
    console.log(`Command received: ${command}`);
  }
}
```

## Auto-Reply System

Build an intelligent auto-reply system:

```javascript
class AutoReplySystem {
  constructor(apiUrl) {
    this.apiUrl = apiUrl;
    this.rules = [];
    this.contexts = new Map(); // Store conversation contexts
  }
  
  addRule(pattern, response, options = {}) {
    this.rules.push({
      pattern: new RegExp(pattern, 'i'),
      response,
      ...options
    });
  }
  
  async handleMessage(message) {
    if (message.fromMe) return; // Don't reply to own messages
    
    const userContext = this.getContext(message.from);
    
    // Check for rules
    for (const rule of this.rules) {
      if (rule.pattern.test(message.body)) {
        let response = rule.response;
        
        // Dynamic response generation
        if (typeof response === 'function') {
          response = await response(message, userContext);
        }
        
        // Template replacement
        response = this.replaceTemplates(response, message, userContext);
        
        await this.sendReply(message.from, response);
        
        // Update context if needed
        if (rule.updateContext) {
          this.updateContext(message.from, rule.updateContext);
        }
        
        break; // Stop after first match
      }
    }
  }
  
  getContext(userJid) {
    if (!this.contexts.has(userJid)) {
      this.contexts.set(userJid, {
        name: null,
        lastActivity: new Date(),
        conversationState: 'initial'
      });
    }
    return this.contexts.get(userJid);
  }
  
  updateContext(userJid, updates) {
    const context = this.getContext(userJid);
    Object.assign(context, updates, { lastActivity: new Date() });
  }
  
  replaceTemplates(text, message, context) {
    return text
      .replace('{name}', context.name || 'there')
      .replace('{time}', new Date().toLocaleTimeString())
      .replace('{date}', new Date().toLocaleDateString());
  }
  
  async sendReply(phone, message) {
    try {
      const response = await fetch(`${this.apiUrl}/send/message`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ phone, message })
      });
      
      const result = await response.json();
      console.log('Auto-reply sent:', result);
    } catch (error) {
      console.error('Failed to send auto-reply:', error);
    }
  }
}

// Setup auto-reply rules
const autoReply = new AutoReplySystem('http://localhost:3000');

// Basic greetings

);

// Business hours check

  async () => {
    const hour = new Date().getHours();

    } else {
      return 'We are currently closed. Our business hours are 9 AM to 5 PM. We\'ll respond to your message during business hours.';
    }
  }
);

// Name collection

  'Nice to meet you, {name}! How can I assist you?',
  {

  }
```

## Message Queue System

For high-volume applications, implement a message queue:

```javascript
class MessageQueue {
  constructor(options = {}) {
    this.queue = [];
    this.processing = false;
    this.maxConcurrent = options.maxConcurrent || 3;
    this.retryAttempts = options.retryAttempts || 3;
    this.delayBetweenMessages = options.delayBetweenMessages || 1000;
  }
  
  addMessage(message) {
    this.queue.push({
      ...message,
      attempts: 0,
      addedAt: new Date()
    });
    
    if (!this.processing) {
      this.processQueue();
    }
  }
  
  async processQueue() {
    if (this.processing) return;
    
    this.processing = true;
    
    while (this.queue.length > 0) {
      const batch = this.queue.splice(0, this.maxConcurrent);
      
      const promises = batch.map(msg => this.processMessage(msg));
      await Promise.allSettled(promises);
      
      if (this.queue.length > 0) {
        await this.sleep(this.delayBetweenMessages);
      }
    }
    
    this.processing = false;
  }
  
  async processMessage(message) {
    try {
      await this.handleMessage(message);
      console.log(`Message processed: ${message.id}`);
    } catch (error) {
      console.error(`Failed to process message ${message.id}:`, error);
      
      message.attempts++;
      if (message.attempts < this.retryAttempts) {
        console.log(`Retrying message ${message.id}, attempt ${message.attempts}`);
        this.queue.push(message);
      } else {
        console.error(`Message ${message.id} failed after ${this.retryAttempts} attempts`);
      }
    }
  }
  
  async handleMessage(message) {
    // Implement your message handling logic here
    console.log('Processing message:', message);
    
    // Simulate API call delay
    await this.sleep(Math.random() * 1000);
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
  
  getStats() {
    return {
      queueLength: this.queue.length,
      isProcessing: this.processing,
      oldestMessage: this.queue.length > 0 ? this.queue[0].addedAt : null
    };
  }
}

// Usage
const messageQueue = new MessageQueue({
  maxConcurrent: 5,
  retryAttempts: 3,
  delayBetweenMessages: 500
});

// Add messages to queue from webhook
function handleIncomingMessage(message) {
  messageQueue.addMessage(message);
}
```
